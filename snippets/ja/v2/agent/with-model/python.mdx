```python Python
from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

# Spark 2を使用(デフォルトではないため明示的に指定)
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-2"
)

# Spark 1 Miniを使用
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-1-mini"
)

# 複雑なタスクにSpark 1 Pro(デフォルト)を使用
result = app.agent(
    prompt="Compare all enterprise features and pricing across Firecrawl, Apify, and ScrapingBee",
    model="spark-1-pro"
)

print(result.data)
```
