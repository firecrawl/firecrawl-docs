```python Python
from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

result = app.agent(
    prompt="Find recent podcast episodes about AI agents and who appeared on them",
    exchange={
        "enabled": True,
        # Jusqu’à 5 slugs de fournisseurs. Omettez ce champ pour laisser l’agent utiliser tous les fournisseurs.
        "toolkits": ["particle"],
    },
)

print(result.data)
if result.exchange:
    print(result.exchange.paid_calls, result.exchange.credits_used)
```
