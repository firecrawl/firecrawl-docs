```js Node
import { Firecrawl } from 'firecrawl';

const firecrawl = new Firecrawl({ apiKey: "fc-YOUR_API_KEY" });

const result = await firecrawl.agent({
  prompt: "Find recent podcast episodes about AI agents and who appeared on them",
  exchange: {
    enabled: true,
    // Jusqu’à 5 slugs de fournisseurs. Omettez ce champ pour laisser l’agent utiliser tous les fournisseurs.
    toolkits: ["particle"]
  }
});

console.log(result.data);
console.log(result.exchange?.paidCalls, result.exchange?.creditsUsed);
```
