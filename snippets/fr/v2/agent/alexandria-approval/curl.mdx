```bash cURL
# 1. Démarrer une exécution qui demande l’approbation avant tout appel à un fournisseur payant.
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "Find the head of engineering at Firecrawl and a verified work email",
    "mode": "chat",
    "exchange": { "enabled": true, "toolkits": ["apollo"], "requireApproval": true }
  }'

# 2. Interroger l’exécution. Si elle se termine par un pendingApproval, lire son id et ses appels.
curl -X GET "https://api.firecrawl.dev/v2/agent/<jobId>" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY"

# 3. Approuver au tour suivant du même fil, ou envoyer à la place
#    "decline": { "approvalId": "<pendingApproval.id>" }.
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "threadId": "<threadId from the first run>",
    "prompt": "Approved, go ahead.",
    "mode": "chat",
    "exchange": {
      "requireApproval": true,
      "approve": { "approvalId": "<pendingApproval.id>" }
    }
  }'
```
