```python Python
from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

# Utilisation de Spark 2 (pas la valeur par défaut - à définir explicitement)

    prompt="Find the pricing of Firecrawl",
    model="spark-2"
)

# Utilisation de Spark 1 Mini
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-1-mini"
)

# Utilisation de Spark 1 Pro (valeur par défaut) pour les tâches complexes
result = app.agent(
    prompt="Compare all enterprise features and pricing across Firecrawl, Apify, and ScrapingBee",
    model="spark-1-pro"
)

print(result.data)
```
