```bash cURL
# Utilisation de Spark 1 Mini pour réduire les coûts
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "Find the pricing of Firecrawl",
    "model": "spark-1-mini"
  }'

# Utilisation de Spark 1 Pro (par défaut)
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "Comparer toutes les fonctionnalités entreprise et les tarifs de Firecrawl, Apify et ScrapingBee",
    "model": "spark-1-pro"
  }'
```
