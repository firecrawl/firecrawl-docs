```bash cURL
# Utilisation de Spark 2 (par défaut)
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "Trouver les tarifs de Firecrawl",
    "model": "spark-2"
  }'

# Utilisation de Spark 1 Mini

  -d '{
    "prompt": "Trouver les tarifs de Firecrawl",
    "model": "spark-1-mini"
  }'

# Utilisation de Spark 1 Pro pour les tâches complexes
```
