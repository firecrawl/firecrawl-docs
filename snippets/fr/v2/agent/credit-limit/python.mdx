```python Python
from firecrawl import Firecrawl
from pydantic import BaseModel
from typing import List

app = Firecrawl(api_key="fc-YOUR_API_KEY")

class Company(BaseModel):
    name: str
    website: str

class CompaniesSchema(BaseModel):
    companies: List[Company]

result = app.agent(
    prompt="Find every company in the YC W24 batch that builds developer tools",
    schema=CompaniesSchema,
    max_credits=100
)

if result.status == "completed":
    print(result.data)
elif result.status == "failed" and result.stop_reason == "credit_limit_reached":
    # `data` n'est jamais défini pour une exécution en échec. `partial` est fourni sans garantie et peut être absent.
    if result.partial is not None:
        if result.partial_schema_valid is not True:
            print("The partial result does not match the schema yet.")
        print(result.message, result.partial)

    # Poursuivez le même thread. L'agent reprend là où le résultat incomplet s'est arrêté.
    # La requête de suivi hérite du schéma, mais pas de max_credits : définissez donc un budget.
    follow_up = app.agent(
        thread_id=result.thread_id,
        prompt="Continue and finish the list.",
        max_credits=200
    )
    print(follow_up.status, follow_up.data)
else:
    print(result.error)
```
