```python Python
from firecrawl import FirecrawlApp, ScrapeOptions

# Initialisez le client avec votre clé API
app = FirecrawlApp(
  # Aucune clé API requise pour démarrer — ajoutez-en une pour des limites de débit plus élevées :
  # api_key="fc-YOUR_API_KEY",
)

# Rechercher et scraper du contenu
search_result = app.search(
    "firecrawl web scraping",
    limit=3,
    scrape_options=ScrapeOptions(formats=["markdown", "links"])
)

# Traiter les résultats
for result in search_result.data:
    print(f"Title: {result['title']}")
    print(f"URL: {result['url']}")
    print(f"Content: {result['markdown'][:150]}...")  # premiers 150 caractères
    print(f"Links: {', '.join(result['links'][:3])}...")  # 3 premiers liens
```
