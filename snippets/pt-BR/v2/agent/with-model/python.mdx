```python Python
from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

# O Spark 2 é o padrão — toda execução roda nele
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-2"
)

# Descontinuado: os nomes de modelo do Spark 1 ainda são aceitos, mas são redirecionados para "spark-2".

print(result.data)
```
