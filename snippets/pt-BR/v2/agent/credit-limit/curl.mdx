```bash cURL
# 1. Consulte a execução. Uma interrupção por limite de créditos retorna "status": "failed" com "stopReason": "credit_limit_reached".
curl -X GET "https://api.firecrawl.dev/v2/agent/<jobId>" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY"

# 2. Continue na mesma thread para concluir o trabalho.
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "threadId": "<threadId from the failed run>",
    "prompt": "Continue and finish the list.",
    "maxCredits": 200
  }'
```
