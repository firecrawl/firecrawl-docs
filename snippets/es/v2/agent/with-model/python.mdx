```python Python
from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

# Usando Spark 2 (no es el predeterminado - configúralo explícitamente)
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-2"
)

# Usando Spark 1 Mini
result = app.agent(
    prompt="Find the pricing of Firecrawl",
    model="spark-1-mini"
)

# Usando Spark 1 Pro (el predeterminado) para tareas complejas
result = app.agent(
    prompt="Compare all enterprise features and pricing across Firecrawl, Apify, and ScrapingBee",
    model="spark-1-pro"
)

print(result.data)
```
