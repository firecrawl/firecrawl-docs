```bash cURL
# 1. Consulta la ejecución. Una detención por límite de créditos devuelve "status": "failed" con "stopReason": "credit_limit_reached".
curl -X GET "https://api.firecrawl.dev/v2/agent/<jobId>" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY"

# 2. Continúa en el mismo hilo para completar el trabajo.
curl -X POST "https://api.firecrawl.dev/v2/agent" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "threadId": "<threadId from the failed run>",
    "prompt": "Continue and finish the list.",
    "maxCredits": 200
  }'
```
