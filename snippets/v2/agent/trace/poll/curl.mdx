```bash cURL
# Poll the trace, grouping by agent.id before ordering on producerSequence.
while :; do
  curl -s "https://api.firecrawl.dev/v2/agent/JOB_ID/trace" \
    -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
  | jq -r '.events | group_by(.agent.id)[]
           | sort_by(.producerSequence)[]
           | "\(.agent.id) \(.producerSequence) \(.type)"'

  status=$(curl -s "https://api.firecrawl.dev/v2/agent/JOB_ID" \
    -H "Authorization: Bearer $FIRECRAWL_API_KEY" | jq -r '.status')
  [ "$status" = "processing" ] || break
  sleep 5
done
```
