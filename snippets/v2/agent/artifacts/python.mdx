```python Python
import json

from firecrawl import Firecrawl

app = Firecrawl(api_key="fc-YOUR_API_KEY")

trace = app.get_agent_trace("JOB_ID")

for event in trace.events or []:
    if event.type != "artifact.updated":
        continue
    if event.artifact.kind not in ("markdown", "html"):
        continue

    snapshot = app.get_agent_snapshot("JOB_ID", event.artifact.snapshot_id)
    content = json.loads(snapshot.snapshot)  # snapshot is a JSON-encoded string

    print(event.artifact.kind, event.artifact.path, content)
```
