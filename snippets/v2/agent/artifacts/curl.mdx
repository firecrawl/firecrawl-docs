```bash cURL
# Collect the snapshot IDs of every markdown or html artifact the run wrote.
curl -s "https://api.firecrawl.dev/v2/agent/JOB_ID/trace" \
  -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
| jq -r '.events[]
         | select(.type == "artifact.updated")
         | select(.artifact.kind == "markdown" or .artifact.kind == "html")
         | .artifact.snapshotId' \
| while read -r snapshot_id; do
    curl -s "https://api.firecrawl.dev/v2/agent/JOB_ID/snapshots/$snapshot_id" \
      -H "Authorization: Bearer $FIRECRAWL_API_KEY" \
    | jq -r '.snapshot | fromjson'   # snapshot is a JSON-encoded string
  done
```
